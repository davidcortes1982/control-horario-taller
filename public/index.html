function enviarFichajeConGPS(email, password, naveId, tipoFichaje) { // Comprobar si el navegador permite geolocalización if (!navigator.geolocation) { alert("Tu navegador no soporta geolocalización."); return; } // Pedir la posición exacta al móvil navigator.geolocation.getCurrentPosition( (position) => { const lat = position.coords.latitude; const lon = position.coords.longitude; // Enviar los datos junto con la latitud y longitud al servidor fetch('/api/fichar', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email: email, password: password, nave_id: naveId, tipo: tipoFichaje, lat: lat, lon: lon }) }) .then(res => res.json()) .then(data => { if (data.exito) { alert(`¡Fichaje exitoso a las ${data.hora}!`); } else { alert(`Error: ${data.error}`); } }) .catch(err => alert("Error de conexión con el servidor.")); }, (error) => { alert("Debes activar y permitir el acceso al GPS para poder fichar."); }, { enableHighAccuracy: true, timeout: 10000 } ); }
